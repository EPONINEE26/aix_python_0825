    while True:
        str = f.readline()
        if str == "": break
        stu = str.split(",") #,기준으로 리스트생성
        for i,s in enumerate(stu): # 1,홍길동,100,100,100,300,100.0
            if 0<=i<=1: continue
            elif 2<=i<=5:
                stu[i] = int(s.strip())  # s[i] = 문자열 1글자
            elif i==6:
                stu[i] = float(s.strip()) # /n

        stuList.append(stu)
