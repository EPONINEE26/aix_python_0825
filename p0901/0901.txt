import random
a_arr = list(range(1,26))
random.shuffle(a_arr)
while True:
    print(" "*15,end="")
    print("[ 빙고게임 ]")
    print("-"*50)
    for i,v in enumerate(a_arr):
        if (i+1)%5!=0:
            print(v,end="\t")
        else:
            print(v)
    print("-"*50)
    num = int(input("원하는 번호를 입력하세요.>> "))
    if num in a_arr:
        idx = a_arr.index(num)
        a_arr[idx] = "X"
